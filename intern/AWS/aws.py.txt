import json

def lambda_handler(event, context):
    # Extract numbers from the event
    try:
        num1 = event.get('num1')
        num2 = event.get('num2')
        
        # Check if the numbers are provided and are of type int or float
        if num1 is None or num2 is None:
            return {
                'statusCode': 400,
                'body': json.dumps('Error: Please provide both num1 and num2.')
            }
        
        if not isinstance(num1, (int, float)) or not isinstance(num2, (int, float)):
            return {
                'statusCode': 400,
                'body': json.dumps('Error: Both num1 and num2 must be numbers.')
            }
        
        # Calculate the sum
        result = num1 + num2
        
        # Return the result
        return {
            'statusCode': 200,
            'body': json.dumps({'result': result})
        }
    
    except Exception as e:
        return {
            'statusCode': 500,
            'body': json.dumps(f'Error: {str(e)}')
        }
